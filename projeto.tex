% \revisar{texto}

%                     validados pelo orientador/corretor.
%       - Formatação: Exibe o texto em verde escuro.
%       - Exemplo: \revisar{Esta nova metodologia foi aplicada para resolver o problema X.}

% \revisar{texto}
